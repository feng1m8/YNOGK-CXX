program xicelke
!
! test driver of routines "icelk" and "icele"
!
integer jend,j
real*16 dqmc,qmc,qm,qcel,qk,qec
real*8 EPS,mc,m,k,ec,mk,me,dmk,dme
real*8 icelk,icele

EPS=1.11d-16

write(*,'(1x,a)') 'Invert Complete Elliptic Integrals of First and Second Kind'
write(*,'(1x,a10,2a25,2a10)') 'm','K','E','dmk','dme'

jend=20
dqmc=1.q0/qext(jend)
do j=1,jend
    qmc=qext(j)*dqmc
    qm=1.q0-qmc
    mc=dble(qmc)
	m=1.d0-mc
    qk=qcel(sqrt(qmc),1.q0,1.q0,1.q0,ierr)
    qec=qcel(sqrt(qmc),1.q0,1.q0,qmc,ierr)-1.q0
    k=dble(qk)
    ec=dble(qec)
    mk=icelk(k)
    me=icele(ec)
    dmk=mk-m
    dme=me-m
	write(*,'(1x,0p1f10.7,0p2f25.16,1p2e10.2)') m,k,ec,dmk,dme
enddo

end program xicelke

      real*16 function qcel(kc0, nc, aa, bb, err)
c
c	quad precision general complete elliptic integral "cel"
c
c	created by Burlisch
c	coded by T. Fukushima <Toshio.Fukushima@nao.ac.jp>
c
c	Reference: Fukushima, T, and Ishizaki, H. (1994)
c		qcelest. Mech. Dyn. Astron.	59,	237-251
c		"Numerical Computation of Incomplete Elliptic Integrals of a General Form"
c
      integer err
      real*16 kc0, nc, aa, bb
      real*16 out, a, b, e, f, g
      real*16 p, q, kc, em, qc
      err = 0
      kc = kc0
      if (kc .ne. 0.q0) goto 3
         if (b .ne. 0.q0) goto 1
            kc = 9.27q-69
            goto  2
   1        err = 1
            qcel = 1.q99
            return
   2     continue
   3  qc = abs(kc)
      a = aa
      b = bb
      p = nc
      e = qc
      em = 1.q0
      if (p .le. 0.q0) goto 4
         p = sqrt(p)
         b = b/p
         goto  5
   4     f = qc*qc
         q = 1.q0-f
         g = 1.q0-p
         f = f-p
         q = q*(b-a*p)
         p = sqrt(f/g)
         a = (a-b)/g
         b = a*p-q/(g*g*p)
   5  continue
         f = a
         a = a+b/p
         g = e/p
         b = b+f*g
         b = 2.q0*b
         p = p+g
         g = em
         em = em+qc
         if (abs(g-qc) .lt. 9.8q-18*qc) goto  6
         qc = 2.q0*sqrt(e)
         e = qc*em
         goto  5
   6  out = 1.57079632679489661923132169163975q0
     &      *(b+a*em)/(em*(em+p))
      qcel = out
      return
      end
