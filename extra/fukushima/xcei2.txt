!===============================================================================
!  This file contains the single and double precision Fortran 90 functions
!  to compute five complete elliptic integrals: K(m), E(m), B(m), D(m), and S(m)
!
!     1) xcei2.f90  test driver of "sceik", "sceie", "sceib", "sceid", "sceis",
!        "ceik", "ceie", "ceib", "ceid", and "ceis"
!     2) sceik.f90: single precision function to compute K(m)
!     3) sceie.f90: single precision function to compute E(m)
!     4) sceib.f90: single precision function to compute B(m) = (E(m)-(1-m)K(m))/m
!     5) sceid.f90: single precision function to compute D(m) = (K(m)-E(m))/m
!     6) sceis.f90: single precision function to compute S(m) = (D(m)-B(m))/m
!     7) ceik.f90: double precision function to compute K(m)
!     8) ceie.f90: double precision function to compute E(m)
!     9) ceib.f90: double precision function to compute B(m)
!     10) ceid.f90: double precision function to compute D(m)
!     11) ceis.f90: double precision function to compute S(m)
!
!  Notice that B(m) and S(m) are most fundamental since the others are computable
!  from them without suffering precision loss even when m is small such that
!     D(m) = B(m) + m S(m)
!     K(m) = B(m) + D(m) = 2 B(m) + m S(m)
!     E(m) = m B(m) + (1-m) K(m) = (2-m) B(m) + m (1-m) S(m)
!
!  The file also includes the output file of "xcei2" 
!
!===============================================================================
program xcei2
!
! Test driver of the single and double precision functions,
! "sceik", "sceie", "sceib","sceid", "sceis", "ceik", "ceie", "ceib","ceid", and "ceis"
! to compute the five complete elliptic integrals of the first and second kinds,
! K(m), E(m), B(m)=(E(m)-(1-m)K(m))/m, D(m)=(K(m)-E(m))/m, S(m)=((2-m)K(m)-2E(m))/m^2
!
! References:
!
! T. Fukushima, (2015), J. Comp. Appl. Math., 282, 71-76
!
! "Precise and Fast Computation of General Complete Elliptic
! Integral of Second Kind"
!
! T. Fukushima, (2016), Astron. J.,	152, 35
!
! "Zonal Toroidal Harmonic Expansions of External Gravitational
! Fields for Ring-Like Objects"
!
! Author: T. Fukushima Toshio.Fukushima@nao.ac.jp
!
implicit integer (i-n)
implicit real*8 (a-h,o-z)
!
kend=40
dm=1.d0/dble(kend)
write (*,"(a10,a100,a65)") "Parameter","Double Precision Approximation", &
    "Single Precision Approximation"
write (*,"(a10,5a20,5a13)") "m","K(m)","E(m)","B(m)","D(m)","S(m)", &
    "K(m)","E(m)","B(m)","D(m)","S(m)"
do k=0,kend-1
    fm=dm*dble(k)
    fmc=1.d0-fm
    fK=ceik(fmc)
    fE=ceie(fmc)
    fB=ceib(fmc)
    fD=ceid(fmc)
    fS=ceis(fmc)
    sK=sceik(fmc)
    sE=sceie(fmc)
    sB=sceib(fmc)
    sD=sceid(fmc)
    sS=sceis(fmc)
    write (*,"(0pf10.5,0p5f20.15,0p5f13.8)") fm,fK,fE,fB,fD,fS,sK,sE,sB,sD,sS
enddo
fm=1.d0
fmc=0.d0
fE=ceie(fmc)
fB=ceib(fmc)
sE=sceie(fmc)
sB=sceib(fmc)
write (*,"(0pf10.5,20x,0p2f20.15,53x,0p2f13.8)") fm,fE,fB,sE,sB
!
end program xcei2
