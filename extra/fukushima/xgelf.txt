program xgelf
!
! test driver for gelf
!
implicit none
real*8 PI,PIHALF
parameter (PI=3.1415926535897932384626433d0)
parameter (PIHALF=PI*0.5d0)
real*8 dmc,mc,m,dphi,phi,f,gelf
real*4 rm,rphi,rf,rgelf,rdf
integer jend,iend,j,i
!
jend=10
iend=8
!
dmc=1.d0/dble(jend)
write(*,'(1x,2a10,a25,a10)') "m","phi","F","rF-F"
!
do j=1,jend
    mc=dble(j)*dmc
	m=1.d0-mc
    rm=m
	dphi=PIHALF/dble(iend)
    do i=0,iend*8
	    phi=dphi*dble(i)
        rphi=phi
	    f=gelf(phi,m)
	    rf=rgelf(rphi,rm)
        rdf=(rf-f)/(1.d-16+abs(f))
        write(*,'(1x,0p2f10.5,1pe25.15,1pe10.2)') m,phi,f,rdf
	enddo
    write(*,'(1x)')
enddo
end program xgelf

real*8 function gelf(phi,m)
!
!	Double precision incomplete elliptic integral of the fist kind
!
!   For general amplitude: -infty < phi < infty
!
!     Reference: T. Fukushima, (2010) Numer. Math. 116, 687-719
!        "Fast Computation of Incomplete Elliptic Integral of First Kind
!         by Half Argument Transformation"		
!
!     Author: T. Fukushima Toshio.Fukushima@nao.ac.jp
!
!     Used subprograms: elf, elk
!
!     Inputs: phi = argument, m = complementary parameter, 0 <= m < 1
!
!     Output: gelf
!
real*8 phi,m
real*8 elf,elk
real*8 mc,phix,k,phic,f
real*8 PIHALF,PI,TWOPI,PIH3
integer n
parameter (PI=3.141592653589793238462d0)
parameter (PIHALF=PI*0.5d0,TWOPI=PI*2.d0,PIH3=PI*1.5d0)
!
if(m.ge.1.d0) then
	write(*,*) "(gelf) too large parameter: m=",m
    return
endif
mc=1.d0-m
phix=abs(phi)
k=elk(mc)
n=int(phix/TWOPI)
phix=phix-dble(n)*TWOPI
if(phix.lt.PIHALF) then
    phic=PIHALF-phix; f=elf(phix,phic,mc)
elseif(phix.lt.PI) then
    phix=PI-phix
    phic=PIHALF-phix; f=elf(phix,phic,mc)
    f=2.d0*k-f
elseif(phix.lt.PIH3) then
    phix=phix-PI
    phic=PIHALF-phix; f=elf(phix,phic,mc)
    f=f+2.d0*k
else
    phix=TWOPI-phix
    phic=PIHALF-phix; f=elf(phix,phic,mc)
    f=4.d0*k-f
endif
if(n.gt.0) f=f+dble(4*n)*k
if(phi.lt.0.d0) f=-f
gelf=f
return
end

real*4 function rgelf(phi,m)
!
!	Single precision incomplete elliptic integral of the fist kind
!
!     Reference: T. Fukushima, (2010) Numer. Math. 116, 687-719
!        "Fast Computation of Incomplete Elliptic Integral of First Kind
!         by Half Argument Transformation"		
!
!     Author: T. Fukushima Toshio.Fukushima@nao.ac.jp
!
!     Used subprograms: relf, rasn, racn, relk, rserf
!
!     Inputs: phi = argument, m = complementary parameter, 0 <= m < 1
!
!     Output: rgelf
!
real*4 phi,m
real*4 relf,relk
real*4 phix,mc,k,phic,f
real*4 PIHALF,PI,TWOPI,PIH3
integer n
parameter (PI=3.141592653589793238462)
parameter (PIHALF=PI*0.5,TWOPI=PI*2.0,PIH3=PI*1.5)
if(m.ge.1.0) then
	write(*,*) "(rgelf) too large parameter: m=",m
    return
endif
mc=1.0-m
phix=abs(phi)
k=relk(mc)
n=int(phix/TWOPI)
phix=phix-real(n)*TWOPI
if(phix.lt.PIHALF) then
    phic=PIHALF-phix; f=relf(phix,phic,mc)
elseif(phix.lt.PI) then
    phix=PI-phix
    phic=PIHALF-phix; f=relf(phix,phic,mc)
    f=2.0*k-f
elseif(phix.lt.PIH3) then
    phix=phix-PI
    phic=PIHALF-phix; f=relf(phix,phic,mc)
    f=f+2.0*k
else
    phix=TWOPI-phix
    phic=PIHALF-phix; f=relf(phix,phic,mc)
    f=4.0*k-f
endif
if(n.gt.0) f=f+real(4*n)*k
if(phi.lt.0.0) f=-f
rgelf=f
return
end

      real*8 function elf(phi,phic,mc)
c
c	Double precision incomplete elliptic integral of the first kind
c
c     For limited range of amplitude 0 <= phi <= pi/2
c
c     Reference: T. Fukushima, (2010) Numer. Math. 116, 687-719
c        "Fast Computation of Incomplete Elliptic Integral of First Kind
c         by Half Argument Transformation"		
c
c     Author: T. Fukushima Toshio.Fukushima@nao.ac.jp
c
c     Used subprograms: asn, acn, elk, serf (called from asn)
c
c     Inputs: phi  = argument                0 <= phi  <= PI/2
c             phic = complementar argument   0 <= phic <= PI/2
c             mc   = complementary parameter 0 <  mc   <= 1
c
c     Output: elf
c
c     CAUTION: phi and phic must satisfy condition, phi + phic = PI/2
c
      real*8 phi,phic,mc
	real*8 asn,elk,acn
      real*8 m,c,yc,d2,v
c
	m=1.d0-mc
	if(phi.lt.1.25d0) then
		elf=asn(sin(phi),m)
	else
		c=sin(phic)
		yc=c*c
		d2=mc+m*yc
		if(yc.lt.0.9d0*d2) then
			elf=elk(mc)-asn(c/sqrt(d2),m)
		else
			v=mc*(1.d0-yc)
			if(v.lt.yc*d2) then
				elf=acn(c,mc)
			else
				elf=elk(mc)-asn(sqrt(v/d2),m)
			endif
		endif
	endif
	return
      end
cccccccccccccccccccccccccccccccccccccccccccccccccccccccccc
      real function relf(phi,phic,mc)
c
c	Single precision incomplete elliptic integral of the fist kind
c
c     Reference: T. Fukushima, (2010) Numer. Math. 116, 687-719
c        "Fast Computation of Incomplete Elliptic Integral of First Kind
c         by Half Argument Transformation"		
c
c     Author: T. Fukushima Toshio.Fukushima@nao.ac.jp
c
c     Used subprograms: rasn, racn, relk, rserf (called from rasn)
c
c     Inputs: phi  = argument                0 <= phi  <= PI/2
c             phic = complementar argument   0 <= phic <= PI/2
c             mc   = complementary parameter 0 <= mc   <= 1
c
c     Output: relf
c
c     CAUTION: phi and phic must satisfy condition, phi + phic = PI/2
c
      real phi,phic,mc
	real rasn,relk,racn
      real m,c,yc,d2,v
c
	m=1.0-mc
	if(phi.lt.1.25) then
		relf=rasn(sin(phi),m)
	else
		c=sin(phic)
		yc=c*c
		d2=mc+m*yc
		if(yc.lt.0.9*d2) then
			relf=relk(mc)-rasn(c/sqrt(d2),m)
		else
			v=mc*(1.0-yc)
			if(v.lt.yc*d2) then
				relf=racn(c,mc)
			else
				relf=relk(mc)-rasn(sqrt(v/d2),m)
			endif
		endif
	endif
	return
      end
ccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccc
      real*8 function acn(c0,mc)
c
c	Double precision function to compute inverse cosine amplitude function
c
c     Reference: T. Fukushima, (2010) Numer. Math. 116, 687-719
c        "Fast Computation of Incomplete Elliptic Integral of First Kind
c         by Half Argument Transformation"		
c
c     Author: T. Fukushima Toshio.Fukushima@nao.ac.jp
c
c     Used subprograms: asn
c
c     Inputs: c0  = cosine amplitude function value
c             mc   = complementary parameter 0 <  mc   <= 1
c
c     Output: acn
c
      real*8 c0,mc
	real*8 m,c,yc,y,s,f,d
	real*8 asn
      integer i
	m=1.d0-mc
	c=c0
	yc=c*c
	if(yc.gt.0.5d0) then
		y=1.d0-yc
		s=sqrt(y)
		acn=asn(s,m)
		return
	endif
	f=1.d0
      do i=1,20
    		d=sqrt(mc+m*yc)
		yc=(c+d)/(1.d0+d)
		f=f*2.d0
		if(yc.gt.0.5d0) then
			y=1.d0-yc
			s=sqrt(y)
			goto 1
		endif
		c=sqrt(yc)
	enddo
	write(*,*) "(acn) too many iterations: c0,mc=",c0,mc
    1 continue
	acn=f*asn(s,m)
	return
      end
cccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccc
      real*8 function asn(s0,m)
c
c	Double precision function to compute inverse sine amplitude function
c
c     Reference: T. Fukushima, (2010) Numer. Math. 116, 687-719
c        "Fast Computation of Incomplete Elliptic Integral of First Kind
c         by Half Argument Transformation"		
c
c     Author: T. Fukushima Toshio.Fukushima@nao.ac.jp
c
c     Used subprograms: serf
c
c     Inputs: s0  = sine amplitude function value
c             mc  = complementary parameter 0 <  mc   <= 1
c
c     Output: acn
c
      real*8 s0,m
      real*8 del,s,f,y
	real*8 serf
	integer j
	del=0.04094d0-0.00652d0*m
	s=s0
	y=s*s
	if(y.lt.del) then
		asn=s*serf(y,m)
		return
	endif
	f=1.d0
      do j=1,20
    		y=y/((1.d0+sqrt(1.d0-y))*(1.d0+sqrt(1.d0-m*y)))
		f=f*2.d0
		if(y.lt.del) then
			s=sqrt(y)
			goto 1
		endif
	enddo
	write(*,*) "(asn) too many iterations: s0,m=",s0,m
    1 continue
	asn=f*s*serf(y,m)
	return
      end
cccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccc
      real*8 function serf(y,m)
c
c	Double precision function to compute series expansion
c          required in inverse sine amplitude function
c
c     Reference: T. Fukushima, (2010) Numer. Math. 116, 687-719
c        "Fast Computation of Incomplete Elliptic Integral of First Kind
c         by Half Argument Transformation"		
c
c     Author: T. Fukushima Toshio.Fukushima@nao.ac.jp
c
c     Inputs: y  = square of sine amplitude function value
c             m  = parameter 0 <=  m   < 1
c
c     Output: serf
c
      real*8 y,m
c
	real*8 F1,F2,F3,F4
	real*8 F10,F20,F21,F30,F31,F40,F41,F42
	real*8 F5,F50,F51,F52,F6,F60,F61,F62,F63
	real*8 F7,F70,F71,F72,F73,F8,F80,F81,F82,F83,F84
	real*8 F9,F90,F91,F92,F93,F94
	parameter (F10=1.d0/6.d0)
	parameter (F20=3.d0/40.d0)
	parameter (F21=2.d0/40.d0)
	parameter (F30=5.d0/112.d0)
	parameter (F31=3.d0/112.d0)
	parameter (F40=35.d0/1152.d0)
	parameter (F41=20.d0/1152.d0)
	parameter (F42=18.d0/1152.d0)
	parameter (F50=63.d0/2816.d0)
	parameter (F51=35.d0/2816.d0)
	parameter (F52=30.d0/2816.d0)
	parameter (F60=231.d0/13312.d0)
	parameter (F61=126.d0/13312.d0)
	parameter (F62=105.d0/13312.d0)
	parameter (F63=100.d0/13312.d0)
	parameter (F70=429.d0/30720.d0)
	parameter (F71=231.d0/30720.d0)
	parameter (F72=189.d0/30720.d0)
	parameter (F73=175.d0/30720.d0)
	parameter (F80=6435.d0/557056.d0)
	parameter (F81=3432.d0/557056.d0)
	parameter (F82=2722.d0/557056.d0)
	parameter (F83=2520.d0/557056.d0)
	parameter (F84=2450.d0/557056.d0)
	parameter (F90=12155.d0/1245184.d0)
	parameter (F91=6435.d0/1245184.d0)
	parameter (F92=5148.d0/1245184.d0)
	parameter (F93=4620.d0/1245184.d0)
	parameter (F94=4410.d0/1245184.d0)
	F1=F10+m*F10
	F2=F20+m*(F21+m*F20)
	F3=F30+m*(F31+m*(F31+m*F30))
	F4=F40+m*(F41+m*(F42+m*(F41+m*F40)))
	F5=F50+m*(F51+m*(F52+m*(F52+m*(F51+m*F50))))
	F6=F60+m*(F61+m*(F62+m*(F63+m*(F62+m*(F61+m*F60)))))
	F7=F70+m*(F71+m*(F72+m*(F73+m*(F73+m*(F72+m*(F71+m*F70))))))
	F8=F80+m*(F81+m*(F82+m*(F83+m*(F84
     &	+m*(F83+m*(F82+m*(F81+m*F80)))))))
	F9=F90+m*(F91+m*(F92+m*(F93+m*(F94+m*(F94
     &	+m*(F93+m*(F92+m*(F91+m*F90))))))))
	serf=1.d0+y*(F1+y*(F2+y*(F3+y*(F4
     &	+y*(F5+y*(F6+y*(F7+y*(F8+y*F9))))))))
	return
      end

ccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccc
      real function racn(c0,mc)
c
c	Single precision function to compute inverse cosine amplitude function
c
c     Reference: T. Fukushima, (2010) Numer. Math. 116, 687-719
c        "Fast Computation of Incomplete Elliptic Integral of First Kind
c         by Half Argument Transformation"		
c
c     Author: T. Fukushima Toshio.Fukushima@nao.ac.jp
c
c     Used subprograms: rasn
c
c     Inputs: c0  = cosine amplitude function value
c             mc   = complementary parameter 0 <  mc   <= 1
c
c     Output: racn
c
      real c0,mc
	real m,c,yc,y,s,f,d
	real rasn
      integer i
	m=1.0-mc
	c=c0
	yc=c*c
	if(yc.gt.0.5) then
		y=1.0-yc
		s=sqrt(y)
		racn=rasn(s,m)
		return
	endif
	f=1.0
      do i=1,10
    		d=sqrt(mc+m*yc)
		yc=(c+d)/(1.0+d)
		f=f*2.0
		if(yc.gt.0.5) then
			y=1.0-yc
			s=sqrt(y)
			goto 1
		endif
		c=sqrt(yc)
	enddo
	write(*,*) "(racn) too many iterations: c0,mc=",c0,mc
    1 continue
	racn=f*rasn(s,m)
	return
      end
ccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccc
      real function rasn(s0,m)
c
c	Single precision function to compute inverse sine amplitude function
c
c     Reference: T. Fukushima, (2010) Numer. Math. 116, 687-719
c        "Fast Computation of Incomplete Elliptic Integral of First Kind
c         by Half Argument Transformation"		
c
c     Author: T. Fukushima Toshio.Fukushima@nao.ac.jp
c
c     Used subprograms: rserf
c
c     Inputs: s0  = sine amplitude function value
c             mc  = complementary parameter 0 <  mc   <= 1
c
c     Output: racn
c
      real s0,m
      real del,s,f,y
	real rserf
	integer j
	integer jsn,jcn
	common /count/ jsn,jcn
	del=0.1888-0.0378*m
	s=s0
	y=s*s
	if(y.lt.del) then
		rasn=s*rserf(y,m)
		jsn=0
		return
	endif
	f=1.0
      do j=1,10
    		y=y/((1.0+sqrt(1.0-y))*(1.0+sqrt(1.0-m*y)))
		f=f*2.0
		if(y.lt.del) then
			s=sqrt(y)
			jsn=j
			goto 1
		endif
	enddo
	write(*,*) "(rasn) too many iterations: s0,m=",s0,m
    1 continue
	rasn=f*s*rserf(y,m)
	return
      end

      real function rserf(y,m)
c
c	Single precision function to compute series expansion
c          required in inverse sine amplitude function
c
c     Reference: T. Fukushima, (2010) Numer. Math. 116, 687-719
c        "Fast Computation of Incomplete Elliptic Integral of First Kind
c         by Half Argument Transformation"		
c
c     Author: T. Fukushima Toshio.Fukushima@nao.ac.jp
c
c     Inputs: y  = square of sine amplitude function value
c             m  = parameter 0 <=  m   < 1
c
c     Output: rserf
c
      real y,m
c
	real F1,F2,F3,F4
	real F10,F20,F21,F30,F31,F40,F41,F42
	real F5,F50,F51,F52,F6,F60,F61,F62,F63
	parameter (F10=1.0/6.0)
	parameter (F20=3.0/40.0)
	parameter (F21=2.0/40.0)
	parameter (F30=5.0/112.0)
	parameter (F31=3.0/112.0)
	parameter (F40=35.0/1152.0)
	parameter (F41=20.0/1152.0)
	parameter (F42=18.0/1152.0)
	parameter (F50=63.0/2816.0)
	parameter (F51=35.0/2816.0)
	parameter (F52=30.0/2816.0)
	parameter (F60=231.0/13312.0)
	parameter (F61=126.0/13312.0)
	parameter (F62=105.0/13312.0)
	parameter (F63=100.0/13312.0)
	F1=F10+m*F10
	F2=F20+m*(F21+m*F20)
	F3=F30+m*(F31+m*(F31+m*F30))
	F4=F40+m*(F41+m*(F42+m*(F41+m*F40)))
	F5=F50+m*(F51+m*(F52+m*(F52+m*(F51+m*F50))))
	F6=F60+m*(F61+m*(F62+m*(F63+m*(F62+m*(F61+m*F60)))))
	rserf=1.0+y*(F1+y*(F2+y*(F3+y*(F4
     &	+y*(F5+y*F6)))))
	return
      end
